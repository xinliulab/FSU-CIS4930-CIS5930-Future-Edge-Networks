\begin{frame}{Open RAN is not the same as open-source RAN}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=blue!4,colframe=ranblue,title=Open RAN]
        Published interfaces\\[0.35em]
        Disaggregated functions\\[0.35em]
        Multi-vendor is \emph{possible}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=purple!5,colframe=ranpurple,title=Open source]
        Source code is available\\[0.35em]
        License permits reuse\\[0.35em]
        Interface may stay closed
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \qa{If two boxes claim the same interface, will they work when plugged together?}
     {Not automatically. A specification leaves options, versions, and timing
      choices open. Conformance and interoperability testing --- plugfests, lab
      integration --- is what finishes the job.}
  \source{https://www.o-ran.org/o-ran-resources}{O-RAN ALLIANCE architecture and resources}
\end{frame}

\begin{frame}{Opening the RAN creates both options and work}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=green!5,colframe=softgreen,title=Options]
        \small
        \begin{itemize}\itemsep4pt
          \item More vendor choice
          \item Faster experiments
          \item Programmable control
        \end{itemize}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tcolorbox}[colback=orange!6,colframe=coreorange,title=Work]
        \small
        \begin{itemize}\itemsep4pt
          \item Fronthaul and timing
          \item Integration testing
          \item Bigger security surface
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \qa{Would you rather debug one black box or five open interfaces?}
     {Both are work, in different places. One box hides the failure and hands the
      whole fix to one vendor. Five interfaces make the failure visible, and make
      the integration yours.}
