\begin{frame}{What does the icon actually tell you?}
  \begin{columns}[T,onlytextwidth]
    \column{0.37\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figure/xg/5g_antenna.jpg}
      \source{https://commons.wikimedia.org/wiki/File:Antennes_actives_5G_Ericsson.jpg}{Ericsson active 5G antennas}
    \column{0.59\textwidth}
      \large\textbf{Turn off Wi-Fi for ten seconds.}

      \vspace{0.5em}
      \normalsize
      What appears: LTE, 5G, 5G+, or 5G UW?

      \vspace{0.5em}
      \small
      \begin{itemize}\itemsep4pt
        \item \textbf{LTE:} 4G radio access.
        \item \textbf{5G:} 5G radio is available.
        \item \textbf{5G+:} AT\&T mid- or high-band.
        \item \textbf{5G UW:} Verizon C-band or mmWave.
      \end{itemize}
  \end{columns}

  \glossline{LTE = Long Term Evolution \quad mmWave = millimeter wave}

  \qa{Does that icon describe the radio, the core, or the whole network?}
     {Only the radio access, and even that is filtered through carrier branding.
      It says nothing about the core behind it, and nothing about the speed you
      actually get.}
\end{frame}

\begin{frame}{The first mobile call was basically a flex}
  \begin{columns}[T,onlytextwidth]
    \column{0.34\textwidth}
      \centering
      \href{https://www.weforum.org/videos/the-first-mobile-phone-call-was-made-50-years-ago/}{%
        \includegraphics[width=\linewidth]{figure/xg/1g_cooper_dynatac.jpg}}
      \source{https://commons.wikimedia.org/wiki/File:2007Computex_e21Forum-MartinCooper.jpg}{Martin Cooper with a 1973 DynaTAC prototype}
    \column{0.62\textwidth}
      \large\textbf{Martin Cooper called his rival.}

      \vspace{0.6em}
      \small
      \begin{itemize}\itemsep4pt
        \item Cooper was a Motorola engineer.
        \item In 1973 he called Joel Engel.
        \item Engel led AT\&T's cellular team.
        \item AT\&T was betting on car phones.
        \item Motorola bet on a handheld.
        \item Service reached users in the 1980s.
      \end{itemize}
  \end{columns}

  \vspace{0.4em}
  \takeaway{1G moved the telephone away from the wall.}
\end{frame}

\begin{frame}{The SIM card is the part that knows you}
  \begin{columns}[T,onlytextwidth]
    \column{0.28\textwidth}
      \centering

        \includegraphics[height=0.40\textheight]{figure/xg/2g_nokia3310.jpg}}
      \source{https://commons.wikimedia.org/wiki/File:Nokia_3310.jpg}{Nokia 3310}
    \column{0.22\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figure/xg/sim_card.png}
      \source{https://commons.wikimedia.org/wiki/File:Sim_card.png}{GSM SIM card}
    \column{0.46\textwidth}
      \large\textbf{2G went digital.}

      \vspace{0.4em}
      \small
      \begin{itemize}\itemsep4pt
        \item Digital voice and SMS
        \item Identity moved onto the SIM
        \item Mass-market roaming
      \end{itemize}
  \end{columns}

  \glossline{SMS = Short Message Service \quad SIM = Subscriber Identity Module}

  \qa{Which one is ``you'' to the network: the phone, the SIM, or the number?}
     {The SIM. It stores the subscriber identity and the secret key used to
      authenticate you. The number is only a label the network maps onto that
      identity.}
\end{frame}

\begin{frame}{The 5G icon can lie by omission}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      \centering
      \href{https://www.youtube.com/watch?v=ik15HVNzJVg}{%
        \includegraphics[width=\linewidth]{figure/xg/video_fake5g.jpg}}
      \source{https://www.youtube.com/watch?v=ik15HVNzJVg}{Trying AT\&T's ``5G E'' (YouTube)}
    \column{0.52\textwidth}
      \small
      \begin{itemize}\itemsep5pt
        \item It names a radio, not a speed.
        \item Branding sits on top of it.
        \item Spectrum and congestion decide.
        \item A 5G radio may use a 4G core.
      \end{itemize}
  \end{columns}

  \qa{``Why is LTE faster than 5G here?''}
     {A congested 5G cell on a narrow band can lose to a clean LTE cell. AT\&T's
      old ``5G E'' icon was LTE-Advanced wearing a 5G name.}
  \source{https://www.reddit.com/r/verizon/comments/1qou5p5/lte/}{Paraphrased user question, 2026}
\end{frame}
