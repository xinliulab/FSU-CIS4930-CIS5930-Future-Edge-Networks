\begin{frame}{What does the icon actually tell you?}
  \begin{columns}[T,onlytextwidth]
    \column{0.37\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figure/xg/5g_antenna.jpg}
      \source{https://commons.wikimedia.org/wiki/File:Antennes_actives_5G_Ericsson.jpg}{Ericsson active 5G antennas}
    \column{0.59\textwidth}
      \large\textbf{Turn off Wi-Fi for ten seconds.}

      \vspace{0.5em}
      \normalsize
      What appears: LTE, 5G, 5G+, or 5G UW?

      \vspace{0.5em}
      \small
      \begin{itemize}\itemsep4pt
        \item \textbf{LTE:} 4G radio access.
        \item \textbf{5G:} 5G radio is available.
        \item \textbf{5G+:} AT\&T mid- or high-band.
        \item \textbf{5G UW:} Verizon C-band or mmWave.
      \end{itemize}
  \end{columns}

  \glossline{LTE = Long Term Evolution \quad mmWave = millimeter wave}

  \qa{Does that icon describe the radio, the core, or the whole network?}
     {Only the radio access, and even that is filtered through carrier branding.
      It says nothing about the core behind it, and nothing about the speed you
      actually get.}
\end{frame}

      \vspace{0.4em}
      \small
      \begin{itemize}\itemsep4pt

        \item Identity moved onto the SIM
        \item Mass-market roaming
      \end{itemize}
  \end{columns}

  \glossline{SMS = Short Message Service \quad SIM = Subscriber Identity Module}

  \qa{Which one is ``you'' to the network: the phone, the SIM, or the number?}
     {The SIM. It stores the subscriber identity and the secret key used to
      authenticate you. The number is only a label the network maps onto that
      identity.}
\end{frame}

\begin{frame}{Your phone picks the loudest tower, not the honest one}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.86,transform shape]
        \node[netnode,minimum width=1.35cm] (ph) at (0,0) {Phone};
        \node[netnode,minimum width=1.75cm] (real) at (3.4,1.15) {Real tower};
        \node[oranode,minimum width=1.75cm] (fake) at (3.4,-1.15) {Fake tower};
        \draw[flow,draw=softgray,dashed] (ph) -- (real);
        \node[font=\scriptsize,align=center] at (1.55,1.15) {weak};
        \draw[flow,draw=ranpurple,very thick] (ph) -- (fake);
        \node[font=\scriptsize,align=center] at (1.55,-1.15) {stronger};
      \end{tikzpicture}
    \column{0.46\textwidth}
      \large\textbf{Three steps.}

      \vspace{0.5em}
      \small
      \begin{itemize}\itemsep5pt
        \item The phone attaches to the strongest signal.
        \item The tower asks who it is.
        \item The phone answers with its identity.
      \end{itemize}
  \end{columns}

  \glossline{IMSI = International Mobile Subscriber Identity}

  \takeaway{Equipment built to do this has a name: an IMSI catcher.}
\end{frame}

\begin{frame}{2G checked your ID. You never got to check theirs.}
  \vspace{0.3em}
  \centering
  \begin{tikzpicture}[scale=0.9,transform shape]
    \node[netnode,minimum width=1.8cm] (ph) at (0,0) {Phone};

    \node[font=\small,anchor=south] at (2.6,0.5) {prove who you are};
    \draw[draw=coreorange,very thick,dashed] (4.25,-0.42) -- (0.95,-0.42);
    \node[font=\small,anchor=north] at (2.6,-0.5) {\textcolor{coreorange}{nothing comes back}};
  \end{tikzpicture}

  \vspace{0.9em}
  \qa{The SIM proves the phone is real. What proved the tower was real?}
     {In 2G, nothing. Authentication ran one direction only. Later generations
      add mutual authentication, but the phone still has to pick a tower before
      it can check anything.}

  \vspace{0.2em}
  {\small\textcolor{softgray}{We trace what each generation did about this on Wednesday.}\par}
\end{frame}

\begin{frame}{The 5G icon can lie by omission}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      \centering
      \href{https://www.youtube.com/watch?v=ik15HVNzJVg}{%
        \includegraphics[width=\linewidth]{figure/xg/video_fake5g.jpg}}
      \source{https://www.youtube.com/watch?v=ik15HVNzJVg}{Trying AT\&T's ``5G E'' (YouTube)}
    \column{0.52\textwidth}
      \small
      \begin{itemize}\itemsep5pt
        \item It names a radio, not a speed.
        \item Branding sits on top of it.
        \item Spectrum and congestion decide.
        \item A 5G radio may use a 4G core.
      \end{itemize}
  \end{columns}

  \qa{``Why is LTE faster than 5G here?''}
     {A congested 5G cell on a narrow band can lose to a clean LTE cell. AT\&T's
      old ``5G E'' icon was LTE-Advanced wearing a 5G name.}
  \source{https://www.reddit.com/r/verizon/comments/1qou5p5/lte/}{Paraphrased user question, 2026}
\end{frame}
