% Teaching note (150 s): the ladder, one row per rung. Cover the right columns and
% have the room fill them in. The one sentence to leave with: a phase is useless
% alone; every rung above RSSI is a DIFFERENCE of phases along some axis.
\begin{frame}{The ladder we climbed}
  \vspace{0.4em}
  \centering
  {\footnotesize\setlength{\tabcolsep}{5pt}
  \begin{tabular}{llll}
    \toprule
    Read\ldots & across\ldots & gives & system \\
    \midrule
    \textcolor{softgray}{signal strength} & --- & a room &
      fingerprinting \\[3pt]
    \textcolor{ranblue}{one phase} & --- & nothing (wrapped) & --- \\[3pt]
    \textcolor{ranblue}{phase difference} & \textcolor{ranblue}{antennas} &
      \textbf{angle} & ArrayTrack: 23 cm, 6 APs \\[3pt]
    \textcolor{coreorange}{phase difference} & \textcolor{coreorange}{subcarriers} &
      \textbf{distance} & Chronos: 14 cm, 1 AP \\[3pt]
    \textcolor{softgreen}{both} & both & \textbf{every path} &
      SpotFi: 40 cm, stock APs \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.8em}
  \takeaway{Phase is a 6 cm ruler with no absolute marks.\\
    Everything above RSSI is a \textbf{difference} of phases along some axis.}
\end{frame}
