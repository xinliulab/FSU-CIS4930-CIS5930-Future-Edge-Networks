\begin{frame}{Appendix: the quantities behind every plot}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \small
      \plainterm{Delay $\tau$}{range, $R = c\tau/2$}
      \par\vspace{0.75em}
      \plainterm{Doppler $\nu$}{speed, $v = \nu c / (2 f_c)$}
      \par\vspace{0.75em}
      \plainterm{Bandwidth}{buys range resolution}
      \par\vspace{0.75em}
      \plainterm{Dwell time}{buys velocity resolution}
    \end{column}
    \begin{column}{0.49\textwidth}
      \small
      \plainterm{Rate}{$\sum \log_2(1 + \mathrm{SINR})$ over REs}
      \par\vspace{0.75em}
      \plainterm{Sensing error}{has a hard floor set by bandwidth, power and time}
      \par\vspace{0.75em}
      \plainterm{PSLR}{peak over worst sidelobe, in the RoI}
      \par\vspace{0.75em}
      \textcolor{softgray}{Rate is a sum you maximise. Sensing error has a floor
      you cannot go below.}
    \end{column}
  \end{columns}

  \vspace{0.7em}
  \takeaway{Sensing accuracy is bounded from below.\\That is why the frontier bends.}
\end{frame}
