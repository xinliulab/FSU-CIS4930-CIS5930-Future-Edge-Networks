\begin{frame}{Appendix: the quantities behind every plot}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \small
      \plainterm{Delay $\tau$}{range, $R = c\tau/2$}
      \par\vspace{0.75em}
      \plainterm{Doppler $\nu$}{speed, $v = \nu c / (2 f_c)$}
      \par\vspace{0.75em}
      \plainterm{Bandwidth}{buys range resolution}
      \par\vspace{0.75em}
      \plainterm{Dwell time}{buys velocity resolution}
    \end{column}
    \begin{column}{0.49\textwidth}
      \small
      \plainterm{Rate}{$\sum \log_2(1 + \mathrm{SINR})$ over REs}
      \par\vspace{0.75em}
      \plainterm{CRLB}{inverse of the Fisher information}
      \par\vspace{0.75em}
      \plainterm{PSLR}{peak over worst sidelobe, in the RoI}
      \par\vspace{0.75em}
      \textcolor{softgray}{Rate is a sum you maximise. CRLB is a floor you cannot
      go below.}
    \end{column}
  \end{columns}

  \vspace{0.7em}
  \takeaway{Sensing accuracy is bounded from below.\\That is why the frontier bends.}
\end{frame}

\begin{frame}{Appendix: which waveform for which problem}
  \scriptsize
  \renewcommand{\arraystretch}{1.24}
  \vspace{0.15em}
  \begin{tabularx}{\textwidth}{
      >{\raggedright\arraybackslash}p{1.35cm}
      >{\raggedright\arraybackslash}X
      >{\raggedright\arraybackslash}X
      >{\raggedright\arraybackslash}p{2.2cm}}
    \toprule
    \textbf{Waveform} & \textbf{Best when} & \textbf{Avoid when} & \textbf{Maturity} \\
    \midrule
    OFDM & You must ship on existing 5G hardware & Doppler spread is large &
      Deployed \\
    OTFS & Mobility is high and you want an OFDM front end & Sidelobes must be
      clean & Prototypes \\
    ODDM & Sensing accuracy and multiuser separation both matter & You need it this
      year & Research \\
    DCO-OFDM & The medium is light, not radio & Illumination power is scarce &
      Research \\
    \bottomrule
  \end{tabularx}

  \vspace{0.5em}
  \qa{Why would anyone still choose OFDM for sensing?}
     {Because the alternative costs a new modem in every device. A worse waveform
      that exists beats a better one that does not.}
\end{frame}
