\raggedright
\vspace{8pt}
\begin{columns}[T]
\column{0.46\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item Prefill FLOPs $\propto$ \textbf{active} params.
  \item Bytes read per decode step $\propto$ \textbf{active} params, \emph{if every expert sits in DRAM}.
  \item Memory footprint $\propto$ \textbf{total} params.
\end{itemize}
\column{0.52\textwidth}
\begin{itemize}\setlength{\itemsep}{3pt}
  \item \textbf{Data center}: memory is pooled across GPUs and compute is the bill. MoE wins, so frontier models are MoEs.
  \item \textbf{Phone}: memory is the scarcest resource. Does MoE still pay off?
  \item \alert{MobileMoE (today): yes, at moderate sparsity.} At equal INT4 weight memory it decodes \textbf{2.2--3.4$\times$ faster} than a dense model.
\end{itemize}
\end{columns}
\refline{A.~Q.~Jiang et al., Mixtral of Experts, arXiv:2401.04088, 2024 \textperiodcentered{} DeepSeek-AI, DeepSeek-V3 Technical Report, arXiv:2412.19437, 2024 \textperiodcentered{} Y.~Chen et al., MobileMoE, arXiv:2605.27358, 2026}
\note{21:30--23:00. Connect to Section 1: decode reads only active experts, so MoE directly shrinks bytes per token, as long as the experts are resident.}
\end{frame}

\small
\textbf{DeepSeekMoE} (ACL'24)
\begin{itemize}
  \item Split each expert into $m$ smaller ones and activate $m\times$ as many: sharper specialization.
  \item Isolate a few \emph{shared} experts that every token uses, so routed experts do not duplicate common knowledge.
\end{itemize}
\vspace{4pt}
\textbf{MobileMoE's on-device sweet spot}
\begin{itemize}
  \item Moderate sparsity, fine-grained experts, one shared expert, e.g., \textbf{60 routed + 1 shared, top-4}. Found with a scaling law fitted under phone limits.
\end{itemize}

\end{tikzpicture}
\column{0.56\textwidth}
\small
\begin{itemize}\setlength{\itemsep}{4pt}
  \item \textbf{LLM in a flash} (ACL'24): keep weights on flash and load only what is needed. It reuses recently loaded neurons (``windowing'') and reads larger contiguous chunks (``row--column bundling''). Models up to 2$\times$ the DRAM size run 4--5$\times$ (CPU) and 20--25$\times$ (GPU) faster than with naive loading.
  \item \textbf{The MoE twist}: the router decides which expert it needs only at the last moment. So cache the hot experts in DRAM, and predict and prefetch the rest.
  \item \textbf{D2MoE} (MobiCom'25, on 11/4): route to an expert \emph{and} a precision, and overlap expert loading with compute.
\end{itemize}
\end{columns}
\refline{K.~Alizadeh et al., LLM in a Flash: Efficient Large Language Model Inference with Limited Memory, ACL 2024 \textperiodcentered{} H.~Wang et al., D2MoE, ACM MobiCom 2025}
\note{24:30--26:00. MobileMoE assumes all experts fit; D2MoE drops that assumption. This is the 11/4 hand-off the course page mentions.}
\end{frame}
